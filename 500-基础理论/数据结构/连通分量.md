# 1 重点
- 连通分量是图中相互可达的顶点的最大集合
- 分为强连通分量（有向图）和连通分量（无向图）
- 使用DFS、BFS或并查集算法求解
- 时间复杂度O(V+E)，空间复杂度O(V)
- 广泛应用于网络分析、社交网络、图像处理等领域

# 2 连通分量概念或介绍

## 2.1 基本概念
连通分量（Connected Components）是图论中的一个重要概念，指的是图中相互可达的顶点的最大集合。在无向图中，连通分量是指相互连通的顶点的最大集合；在有向图中，强连通分量是指相互可达的顶点的最大集合。

## 2.2 核心特征
1. **连通性**：同一连通分量内的任意两个顶点都是连通的
2. **最大性**：连通分量是满足连通性的最大顶点集合
3. **互斥性**：不同的连通分量之间没有交集
4. **覆盖性**：所有顶点都属于某个连通分量

## 2.3 数学性质
- **等价关系**：连通性是一种等价关系
- **传递性**：如果A与B连通，B与C连通，则A与C连通
- **自反性**：每个顶点与自身连通
- **对称性**：如果A与B连通，则B与A连通

# 3 连通分量的算法实现

## 3.1 无向图连通分量（DFS）
```python
def find_connected_components_dfs(graph):
    """
    使用DFS查找无向图的连通分量
    """
    def dfs(vertex, component_id):
        visited[vertex] = True
        components[vertex] = component_id
        
        for neighbor in graph.get_neighbors(vertex):
            if not visited[neighbor]:
                dfs(neighbor, component_id)
    
    visited = [False] * graph.vertices
    components = [-1] * graph.vertices
    component_id = 0
    
    for vertex in range(graph.vertices):
        if not visited[vertex]:
            dfs(vertex, component_id)
            component_id += 1
    
    return components, component_id
```

## 3.2 无向图连通分量（BFS）
```python
from collections import deque

def find_connected_components_bfs(graph):
    """
    使用BFS查找无向图的连通分量
    """
    visited = [False] * graph.vertices
    components = [-1] * graph.vertices
    component_id = 0
    
    for vertex in range(graph.vertices):
        if not visited[vertex]:
            # BFS遍历当前连通分量
            queue = deque([vertex])
            visited[vertex] = True
            components[vertex] = component_id
            
            while queue:
                current = queue.popleft()
                
                for neighbor in graph.get_neighbors(current):
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        components[neighbor] = component_id
                        queue.append(neighbor)
            
            component_id += 1
    
    return components, component_id
```

```python
def find_strongly_connected_components_tarjan(graph):
    """
    使用Tarjan算法查找强连通分量
    """
    def tarjan(vertex):
        nonlocal index
        # 设置访问时间和最低可达时间
        dfn[vertex] = low[vertex] = index
        index += 1
        stack.append(vertex)
        in_stack[vertex] = True
        
        for neighbor in graph.get_neighbors(vertex):
            if dfn[neighbor] == -1:  # 未访问
                tarjan(neighbor)
                low[vertex] = min(low[vertex], low[neighbor])
            elif in_stack[neighbor]:  # 在栈中
                low[vertex] = min(low[vertex], dfn[neighbor])
        
        # 如果是最低可达时间等于访问时间，说明是强连通分量的根
        if low[vertex] == dfn[vertex]:
            component = []
            while True:
                top = stack.pop()
                in_stack[top] = False
                component.append(top)
                if top == vertex:
                    break
            scc.append(component)
    
    dfn = [-1] * graph.vertices  # 访问时间
    low = [-1] * graph.vertices  # 最低可达时间
    stack = []
    in_stack = [False] * graph.vertices
    index = 0
    scc = []
    
    for vertex in range(graph.vertices):
        if dfn[vertex] == -1:
            tarjan(vertex)
    
    return scc
```

# 5 连通分量的优化算法

## 5.1 并查集算法
```python
class UnionFind:
    """
    并查集实现
    """
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.count = n
    
    def find(self, x):
        """
        查找根节点（路径压缩）
        """
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        """
        合并两个集合（按秩合并）
        """
        root_x = self.find(x)
        root_y = self.find(y)
        
        if root_x != root_y:
            if self.rank[root_x] < self.rank[root_y]:
                root_x, root_y = root_y, root_x
            
            self.parent[root_y] = root_x
            if self.rank[root_x] == self.rank[root_y]:
                self.rank[root_x] += 1
            self.count -= 1
    
    def get_components(self):
        """
        获取所有连通分量
        """
        components = {}
        for i in range(len(self.parent)):
            root = self.find(i)
            if root not in components:
                components[root] = []
            components[root].append(i)
        return list(components.values())
```

## 5.2 并行连通分量算法
```python
def parallel_connected_components(graph, num_threads=4):
    """
    并行连通分量算法
    """
    import threading
    from concurrent.futures import ThreadPoolExecutor
    
    def process_vertex_range(start, end):
        """
        处理顶点范围
        """
        local_components = []
        visited = [False] * graph.vertices
        
        for vertex in range(start, end):
            if not visited[vertex]:
                component = []
                dfs(vertex, visited, component)
                local_components.append(component)
        
        return local_components
    
    def dfs(vertex, visited, component):
        """
        DFS遍历
        """
        visited[vertex] = True
        component.append(vertex)
        
        for neighbor in graph.get_neighbors(vertex):
            if not visited[neighbor]:
                dfs(neighbor, visited, component)
    
    # 并行处理
    with ThreadPoolExecutor(max_workers=num_threads) as executor:
        chunk_size = graph.vertices // num_threads
        futures = []
        
        for i in range(num_threads):
            start = i * chunk_size
            end = start + chunk_size if i < num_threads - 1 else graph.vertices
            future = executor.submit(process_vertex_range, start, end)
            futures.append(future)
        
        # 合并结果
        all_components = []
        for future in futures:
            all_components.extend(future.result())
    
    return all_components
```

## 6.1 最小生成树

```python
def kruskal_mst(graph):
    """
    使用Kruskal算法求最小生成树
    """
    # 按权重排序边
    edges = []
    for u in range(graph.vertices):
        for v, weight in graph.get_neighbors_with_weights(u):
            if u < v:  # 避免重复边
                edges.append((weight, u, v))
    
    edges.sort()  # 按权重排序
    
    # 使用并查集
    uf = UnionFind(graph.vertices)
    mst = []
    
    for weight, u, v in edges:
        if uf.find(u) != uf.find(v):
            uf.union(u, v)
            mst.append((u, v, weight))
    
    return mst
```

## 6.2 双连通分量
```python
def find_biconnected_components(graph):
    """
    查找双连通分量（无向图）
    """
    def dfs(vertex, parent, depth):
        nonlocal time
        dfn[vertex] = low[vertex] = time
        time += 1
        stack.append(vertex)
        
        for neighbor in graph.get_neighbors(vertex):
            if neighbor == parent:
                continue
            
            if dfn[neighbor] == -1:  # 未访问
                dfs(neighbor, vertex, depth + 1)
                low[vertex] = min(low[vertex], low[neighbor])
                
                # 检查是否是割点
                if low[neighbor] >= dfn[vertex]:
                    component = []
                    while stack[-1] != neighbor:
                        component.append(stack.pop())
                    component.append(stack.pop())
                    component.append(vertex)
                    bcc.append(component)
            else:
                low[vertex] = min(low[vertex], dfn[neighbor])
    
    dfn = [-1] * graph.vertices
    low = [-1] * graph.vertices
    stack = []
    bcc = []
    time = 0
    
    for vertex in range(graph.vertices):
        if dfn[vertex] == -1:
            dfs(vertex, -1, 0)
    
    return bcc
```
