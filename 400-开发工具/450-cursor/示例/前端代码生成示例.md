# 前端代码生成示例

本文档展示如何根据 OpenAPI 规范生成前端代码。

## 给前端 Agent 的提示词

```
根据 docs/api/openapi.yaml 中的用户管理接口规范，生成以下代码：

1. types/user.ts：用户相关的 TypeScript 类型定义
   - User：用户信息类型
   - UserStatus：用户状态枚举
   - CreateUserRequest：创建用户请求类型
   - UpdateUserRequest：更新用户请求类型
   - UserListResponse：用户列表响应类型
   - ApiResponse：统一 API 响应类型

2. api/userApi.ts：用户相关的 API 调用函数（使用 axios）
   - getUserList：获取用户列表
   - createUser：创建用户
   - getUserById：获取用户详情
   - updateUser：更新用户信息
   - deleteUser：删除用户

3. hooks/useUser.ts：React Hook 封装用户 API 调用

要求：
- 完整的 TypeScript 类型定义
- 统一的错误处理
- 支持请求和响应拦截器
- 使用 async/await 语法
- 使用 axios 进行 HTTP 请求
- 类型安全，所有 API 调用都有完整的类型提示
```

## 生成的代码示例

### 1. types/user.ts

```typescript
/**
 * 用户状态枚举
 */
export enum UserStatus {
  ACTIVE = 'ACTIVE',
  INACTIVE = 'INACTIVE',
  DELETED = 'DELETED'
}

/**
 * 用户信息
 */
export interface User {
  /** 用户 ID */
  id: number;
  /** 用户名 */
  name: string;
  /** 邮箱地址 */
  email: string;
  /** 手机号码 */
  phone?: string;
  /** 用户状态 */
  status: UserStatus;
  /** 创建时间 */
  createTime: string;
  /** 更新时间 */
  updateTime?: string;
}

/**
 * 创建用户请求
 */
export interface CreateUserRequest {
  /** 用户名 */
  name: string;
  /** 邮箱地址 */
  email: string;
  /** 密码 */
  password: string;
  /** 手机号码 */
  phone?: string;
}

/**
 * 更新用户请求
 */
export interface UpdateUserRequest {
  /** 用户名 */
  name?: string;
  /** 手机号码 */
  phone?: string;
}

/**
 * 用户列表查询参数
 */
export interface UserListParams {
  /** 页码，从 1 开始 */
  page?: number;
  /** 每页数量 */
  size?: number;
  /** 搜索关键词（匹配用户名或邮箱） */
  keyword?: string;
  /** 用户状态筛选 */
  status?: UserStatus;
}

/**
 * 用户列表数据
 */
export interface UserListData {
  /** 总记录数 */
  total: number;
  /** 当前页码 */
  page: number;
  /** 每页数量 */
  size: number;
  /** 用户列表 */
  items: User[];
}

/**
 * 统一 API 响应格式
 */
export interface ApiResponse<T = any> {
  /** 响应码（200 表示成功） */
  code: number;
  /** 响应消息 */
  message: string;
  /** 响应数据 */
  data: T;
}

/**
 * 用户列表响应
 */
export type UserListResponse = ApiResponse<UserListData>;

/**
 * 用户详情响应
 */
export type UserDetailResponse = ApiResponse<User>;
```

### 2. api/userApi.ts

```typescript
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';
import {
  User,
  UserListParams,
  UserListResponse,
  UserDetailResponse,
  CreateUserRequest,
  UpdateUserRequest,
  ApiResponse
} from '../types/user';

/**
 * 创建 axios 实例
 */
const apiClient: AxiosInstance = axios.create({
  baseURL: '/api',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
});

/**
 * 请求拦截器
 */
apiClient.interceptors.request.use(
  (config) => {
    // 添加认证 token
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

/**
 * 响应拦截器
 */
apiClient.interceptors.response.use(
  (response) => {
    const { data } = response;
    // 如果响应码不是 200，抛出错误
    if (data.code !== 200) {
      return Promise.reject(new Error(data.message || '请求失败'));
    }
    return data;
  },
  (error) => {
    // 统一错误处理
    if (error.response) {
      const { status, data } = error.response;
      switch (status) {
        case 401:
          // 未授权，跳转到登录页
          window.location.href = '/login';
          break;
        case 403:
          return Promise.reject(new Error('没有权限访问'));
        case 404:
          return Promise.reject(new Error('资源不存在'));
        case 500:
          return Promise.reject(new Error('服务器内部错误'));
        default:
          return Promise.reject(new Error(data?.message || '请求失败'));
      }
    }
    return Promise.reject(error);
  }
);

/**
 * 用户 API
 */
export const userApi = {
  /**
   * 获取用户列表
   * @param params 查询参数
   * @returns 用户列表响应
   */
  getUserList: async (params?: UserListParams): Promise<UserListResponse> => {
    const response = await apiClient.get<UserListResponse>('/users', { params });
    return response;
  },

  /**
   * 创建用户
   * @param request 创建用户请求
   * @returns 用户详情响应
   */
  createUser: async (request: CreateUserRequest): Promise<UserDetailResponse> => {
    const response = await apiClient.post<UserDetailResponse>('/users', request);
    return response;
  },

  /**
   * 获取用户详情
   * @param id 用户 ID
   * @returns 用户详情响应
   */
  getUserById: async (id: number): Promise<UserDetailResponse> => {
    const response = await apiClient.get<UserDetailResponse>(`/users/${id}`);
    return response;
  },

  /**
   * 更新用户信息
   * @param id 用户 ID
   * @param request 更新用户请求
   * @returns 用户详情响应
   */
  updateUser: async (id: number, request: UpdateUserRequest): Promise<UserDetailResponse> => {
    const response = await apiClient.put<UserDetailResponse>(`/users/${id}`, request);
    return response;
  },

  /**
   * 删除用户
   * @param id 用户 ID
   * @returns 空响应
   */
  deleteUser: async (id: number): Promise<ApiResponse<void>> => {
    const response = await apiClient.delete<ApiResponse<void>>(`/users/${id}`);
    return response;
  }
};
```

### 3. hooks/useUser.ts

```typescript
import { useState, useCallback } from 'react';
import { userApi } from '../api/userApi';
import {
  User,
  UserListParams,
  UserListResponse,
  CreateUserRequest,
  UpdateUserRequest
} from '../types/user';

/**
 * 用户列表 Hook
 */
export const useUserList = () => {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [data, setData] = useState<UserListResponse | null>(null);

  const fetchUserList = useCallback(async (params?: UserListParams) => {
    setLoading(true);
    setError(null);
    try {
      const response = await userApi.getUserList(params);
      setData(response);
      return response;
    } catch (err) {
      const error = err instanceof Error ? err : new Error('获取用户列表失败');
      setError(error);
      throw error;
    } finally {
      setLoading(false);
    }
  }, []);

  return {
    loading,
    error,
    data,
    fetchUserList
  };
};

/**
 * 用户详情 Hook
 */
export const useUserDetail = () => {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [user, setUser] = useState<User | null>(null);

  const fetchUser = useCallback(async (id: number) => {
    setLoading(true);
    setError(null);
    try {
      const response = await userApi.getUserById(id);
      setUser(response.data);
      return response.data;
    } catch (err) {
      const error = err instanceof Error ? err : new Error('获取用户详情失败');
      setError(error);
      throw error;
    } finally {
      setLoading(false);
    }
  }, []);

  return {
    loading,
    error,
    user,
    fetchUser
  };
};

/**
 * 创建用户 Hook
 */
export const useCreateUser = () => {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  const createUser = useCallback(async (request: CreateUserRequest) => {

    }
  }, []);

  return {
    loading,
    error,
    createUser
  };
};

/**

 */
export const useUpdateUser = () => {
  const [loading, setLoading] = useState(false);

  const updateUser = useCallback(async (id: number, request: UpdateUserRequest) => {

/**
```

### 4. 使用示例

```typescript
import React, { useEffect } from 'react';
import { useUserList, useCreateUser, useDeleteUser } from './hooks/useUser';
import { CreateUserRequest } from './types/user';

const UserListPage: React.FC = () => {
  const { loading, error, data, fetchUserList } = useUserList();
  const { createUser } = useCreateUser();
  const { deleteUser } = useDeleteUser();

  useEffect(() => {
    fetchUserList({ page: 1, size: 10 });
  }, [fetchUserList]);

  const handleCreate = async () => {
    const request: CreateUserRequest = {
      name: '新用户',
      email: 'newuser@example.com',
      password: 'password123'
    };
    try {
      await createUser(request);
      // 刷新列表
      fetchUserList({ page: 1, size: 10 });
    } catch (err) {
      console.error('创建用户失败:', err);
    }
  };

  const handleDelete = async (id: number) => {
    try {
      await deleteUser(id);
      // 刷新列表
      fetchUserList({ page: 1, size: 10 });
    } catch (err) {
      console.error('删除用户失败:', err);
    }
  };

  if (loading) return <div>加载中...</div>;
  if (error) return <div>错误: {error.message}</div>;

  return (
    <div>
      <button onClick={handleCreate}>创建用户</button>
      <table>
        <thead>
          <tr>
            <th>ID</th>
            <th>用户名</th>
            <th>邮箱</th>
            <th>操作</th>
          </tr>
        </thead>
        <tbody>
          {data?.data.items.map((user) => (
            <tr key={user.id}>
              <td>{user.id}</td>
              <td>{user.name}</td>
              <td>{user.email}</td>
              <td>
                <button onClick={() => handleDelete(user.id)}>删除</button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
};

export default UserListPage;
```

## 使用说明

1. **安装依赖**：
   ```bash
   npm install axios
   # 或
   yarn add axios
   ```

2. **配置代理**（开发环境）：
   在 `vite.config.ts` 或 `webpack.config.js` 中配置 API 代理：
   ```typescript
   export default {
     server: {
       proxy: {
         '/api': {
           target: 'http://localhost:8080',
           changeOrigin: true
         }
       }
     }
   }
   ```

3. **使用生成的代码**：
   - 将类型定义文件放在 `src/types/` 目录
   - 将 API 调用文件放在 `src/api/` 目录
   - 将 Hook 文件放在 `src/hooks/` 目录
   - 在组件中导入并使用

4. **类型安全**：
   所有 API 调用都有完整的 TypeScript 类型提示，确保类型安全。
